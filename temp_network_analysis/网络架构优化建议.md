# 网络架构优化与改进建议

## 1. 当前架构的优势

### 1.1 技术优势
- ✅ **类型安全**: Moya枚举定义API，编译时检查，减少运行时错误
- ✅ **统一管理**: HTZNetWorkManager统一处理所有网络请求
- ✅ **容错机制**: 内置Mock数据和错误处理，提高应用稳定性
- ✅ **分层清晰**: View-ViewModel-Service-Network分层明确
- ✅ **缓存支持**: 支持数据缓存和离线访问
- ✅ **日志完整**: 详细的网络请求日志，便于调试

### 1.2 架构优势
- ✅ **可扩展性**: 新增API接口简单，维护成本低
- ✅ **可测试性**: Mock机制便于单元测试和UI测试
- ✅ **可维护性**: 代码结构清晰，职责分离明确
- ✅ **性能优化**: 支持请求合并、缓存策略等优化

## 2. 潜在问题分析

### 2.1 性能问题
- 🔴 **重复请求**: 缺乏请求去重机制，可能导致重复请求
- 🔴 **内存占用**: 大量数据模型常驻内存，可能导致内存峰值
- 🔴 **缓存策略**: 缓存策略较简单，可能导致数据过期或过多
- 🔴 **并发控制**: 缺乏并发请求控制，可能造成网络拥堵

### 2.2 安全问题
- 🔴 **HTTPS支持**: 生产环境需要启用HTTPS
- 🔴 **证书校验**: 缺乏证书固定等安全校验
- 🔴 **请求签名**: 关键接口缺乏请求签名机制
- 🔴 **数据加密**: 敏感数据传输缺乏加密保护

### 2.3 稳定性问题
- 🔴 **错误处理**: 错误处理机制有待完善
- 🔴 **超时机制**: 超时配置较粗略，缺乏智能重试
- 🔴 **网络监听**: 网络状态监听不够及时和准确
- 🔴 **降级策略**: 网络异常时的降级策略需要优化

#### 3.1.1 请求去重机制
```swift
// 扩展 HTZNetWorkManager
private var ongoingRequests: [String: [(Result<JSON, Error>) -> Void]] = [:]

func requestWithDeduplication<T: HandyJSON>(
    _ target: API,
    modelType: T.Type,
    completion: @escaping (Result<T, Error>) -> Void
) {
    let requestKey = "\(target)"
    
    // 检查是否已有相同请求
    if var callbacks = ongoingRequests[requestKey] {
        // 已有请求，添加回调
        callbacks.append({ result in
            switch result {
            case .success(let json):
                if let model = T.deserialize(from: json.rawString()) {
                    completion(.success(model))
                } else {
                    completion(.failure(NetworkError.parseFailed))
                }
            case .failure(let error):
                completion(.failure(error))
            }
        })
        ongoingRequests[requestKey] = callbacks
        return
    }
    
    // 新请求
    ongoingRequests[requestKey] = []
    
    request(target) { [weak self] result in
        guard let self = self else { return }
        
        // 执行所有回调
        if let callbacks = self.ongoingRequests[requestKey] {
            for callback in callbacks {
                callback(result)
            }
        }
        
        // 清理请求记录
        self.ongoingRequests.removeValue(forKey: requestKey)
    }
}
```

#### 3.1.2 智能重试机制
```swift
// 智能重试策略
struct RetryPolicy {
    let maxRetryCount: Int
    let baseDelay: TimeInterval
    let maxDelay: TimeInterval
    let backoffMultiplier: Double
    
    static let `default` = RetryPolicy(
        maxRetryCount: 3,
        baseDelay: 1.0,
        maxDelay: 10.0,
        backoffMultiplier: 2.0
    )
    
    func delay(for attempt: Int) -> TimeInterval {
        let delay = baseDelay * pow(backoffMultiplier, Double(attempt))
        return min(delay, maxDelay)
    }
    
    func shouldRetry(error: Error, attempt: Int) -> Bool {
        guard attempt < maxRetryCount else { return false }
        
        switch error {
        case let urlError as URLError:
            switch urlError.code {
            case .timedOut, .notConnectedToInternet, .networkConnectionLost:
                return true
            default:
                return false
            }
        case _:
            return false
        }
    }
}
```

#### 3.1.3 请求优先级管理

// 请求优先级枚举
enum RequestPriority: Int, Comparable {
    case low = 0
    case normal = 1
    case high = 2
    case urgent = 3
    
    static func < (lhs: RequestPriority, rhs: RequestPriority) -> Bool {
        return lhs.rawValue < rhs.rawValue
    }
}

// 优先级请求管理器
class PriorityRequestManager {
    private var requestQueue: [(API, RequestPriority, () -> Void)] = []
    private var isProcessing = false
    
    func addRequest(_ target: API, priority: RequestPriority, completion: @escaping () -> Void) {
        requestQueue.append((target, priority, completion))
        requestQueue.sort { $0.1 > $1.1 } // 按优先级排序
        processNextRequest()
    }
    
    private func processNextRequest() {
        guard !isProcessing && !requestQueue.isEmpty else { return }
        
        isProcessing = true
        let (target, _, completion) = requestQueue.removeFirst()
        
        // 执行请求
        HTZNetWorkManager.shared.request(target) { _ in
            completion()
            self.isProcessing = false
            self.processNextRequest()
        }
    }
}

// 智能缓存管理器
class SmartCacheManager {
    private let memoryCache = NSCache<NSString, AnyObject>()
    private let diskCache: URL
    private let policy: CachePolicy
    
    init(policy: CachePolicy = .default) {
        self.policy = policy
        self.diskCache = FileManager.default.urls(for: .cachesDirectory, in: .userDomainMask).first!
        self.memoryCache.totalCostLimit = policy.memoryCapacity
    }
    
    func store<T: Codable>(_ object: T, for key: String, strategy: CacheStrategy = .default) {
        switch strategy {
        case .none:
            break
        case .memory:
            storeInMemory(object, for: key)
        case .disk:
            storeOnDisk(object, for: key)
        case .memoryAndDisk:
            storeInMemory(object, for: key)
            storeOnDisk(object, for: key)
        case .custom(let policy):
            // 自定义策略实现
            break
        }
    }
    
    func retrieve<T: Codable>(_ type: T.Type, for key: String, strategy: CacheStrategy = .default) -> T? {
        switch strategy {
        case .none:
            return nil
        case .memory:
            return retrieveFromMemory(type, for: key)
        case .disk:
            return retrieveFromDisk(type, for: key)
        case .memoryAndDisk:
            if let memoryObject = retrieveFromMemory(type, for: key) {
                return memoryObject
            }
            return retrieveFromDisk(type, for: key)
        case .custom(let policy):
            // 自定义策略实现
            return nil
        }
    }
}

#### 3.2.2 数据同步策略
```swift
// 数据同步策略
enum DataSyncStrategy {
    case optimistic   // 乐观更新
    case pessimistic  // 悲观更新
    case eventual     // 最终一致性
}

// 同步管理器
class DataSyncManager {
    private let cacheManager = SmartCacheManager()
    private let syncQueue = DispatchQueue(label: "com.htz.sync", qos: .utility)
    
    func sync<T: Codable>(
        object: T,
        for key: String,
        strategy: DataSyncStrategy = .optimistic
    ) {
        switch strategy {
        case .optimistic:
            // 乐观更新：立即更新缓存，异步同步到服务器
            cacheManager.store(object, for: key, strategy: .memoryAndDisk)
            syncToServer(object, for: key)
            
        case .pessimistic:
            // 悲观更新：先同步到服务器，成功后再更新缓存
            syncToServer(object, for: key) { [weak self] success in
                if success {
                    self?.cacheManager.store(object, for: key, strategy: .memoryAndDisk)
                }
            }
            
        case .eventual:
            // 最终一致性：放入同步队列，按顺序处理
            syncQueue.async {
                self.syncToServer(object, for: key) { success in
                    if success {
                        DispatchQueue.main.async {
                            self.cacheManager.store(object, for: key, strategy: .memoryAndDisk)
                        }
                    }
                }
            }
        }
    }
}
```

// 证书验证委托
class CertificatePinningDelegate: NSObject, URLSessionDelegate {
    func urlSession(_ session: URLSession, didReceive challenge: URLAuthenticationChallenge, completionHandler: @escaping (URLSession.AuthChallengeDisposition, URLCredential?) -> Void) {
        
        guard let serverTrust = challenge.protectionSpace.serverTrust else {
            completionHandler(.cancelAuthenticationChallenge, nil)
            return
        }
        
        // 验证服务器证书
        let policy = SecPolicyCreateSSL(true, nil)
        SecTrustSetPolicies(serverTrust, policy)
        
        var result: SecTrustResultType = .invalid
        let status = SecTrustEvaluate(serverTrust, &result)
        
        if status == errSecSuccess && (result == .unspecified || result == .proceed) {
            completionHandler(.useCredential, URLCredential(trust: serverTrust))
        } else {
            completionHandler(.cancelAuthenticationChallenge, nil)
        }
    }
}

// 请求签名工具
class RequestSigner {
    private let secretKey: String
    
    init(secretKey: String) {
        self.secretKey = secretKey
    }
    
    func signRequest(_ request: URLRequest) -> URLRequest {
        guard let url = request.url,
              let method = request.httpMethod else { return request }
        
        var signedRequest = request
        
        // 构建签名字符串
        var signatureString = "\(method.uppercased())\(url.path)"
        
        // 添加查询参数
        if let components = URLComponents(url: url, resolvingAgainstBaseURL: false),
           let queryItems = components.queryItems {
            let sortedItems = queryItems.sorted { $0.name < $1.name }
            for item in sortedItems {
                if let value = item.value {
                    signatureString += "\(item.name)=\(value)"
                }
            }
        }
        
        // 添加请求体
        if let httpBody = request.httpBody,
           let bodyString = String(data: httpBody, encoding: .utf8) {
            signatureString += bodyString
        }
        
        // 生成签名
        let signature = generateHMACSHA256(signatureString, key: secretKey)
        
        // 添加签名到请求头
        signedRequest.setValue(signature, forHTTPHeaderField: "X-Signature")
        signedRequest.setValue(Date().timeIntervalSince1970.description, forHTTPHeaderField: "X-Timestamp")
        
        return signedRequest
    }
    
    private func generateHMACSHA256(_ string: String, key: String) -> String {
        let keyData = key.data(using: .utf8)!
        let stringData = string.data(using: .utf8)!
        
        var digestData = Data(count: Int(CC_SHA256_DIGEST_LENGTH))
        
        keyData.withUnsafeBytes { keyBytes in
            stringData.withUnsafeBytes { stringBytes in
                CCHmac(CCHmacAlgorithm(kCCHmacAlgSHA256),
                       keyBytes.baseAddress, keyData.count,
                       stringBytes.baseAddress, stringData.count,
                       digestData.withUnsafeMutableBytes { $0 })
            }
        }
        
        return digestData.map { String(format: "%02hhx", $0) }.joined()
    }
}

// 自适应请求管理器
class AdaptiveRequestManager {
    private let performanceMonitor = NetworkPerformanceMonitor.shared
    private var averageResponseTime: Double = 1.0
    private var requestCount = 0
    
    func adaptRequest(_ originalRequest: URLRequest) -> URLRequest {
        var adaptedRequest = originalRequest
        
        // 根据网络状况调整超时时间
        let timeout = calculateOptimalTimeout()
        adaptedRequest.timeoutInterval = timeout
        
        // 根据响应时间调整请求策略
        if averageResponseTime > 2.0 {
            // 响应慢，启用压缩
            adaptedRequest.setValue("gzip", forHTTPHeaderField: "Accept-Encoding")
        }
        
        return adaptedRequest
    }
    
    private func calculateOptimalTimeout() -> TimeInterval {
        let baseTimeout: TimeInterval = 30
        let adaptiveTimeout = min(baseTimeout, averageResponseTime * 3)
        return max(adaptiveTimeout, 10) // 最小10秒
    }
    
    func updatePerformanceMetrics(duration: TimeInterval) {
        requestCount += 1
        averageResponseTime = (averageResponseTime * Double(requestCount - 1) + duration) / Double(requestCount)
    }
}
