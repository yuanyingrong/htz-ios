## 1. 网络管理核心类

### HTZNetWorkManager.swift
```swift
// 文件位置: htz/htz/Utils/HTZNetWorkManager.swift
// 功能: 统一网络请求管理，提供容错机制

import Moya
import SwiftyJSON
import Alamofire

class HTZNetWorkManager {
    
    // MARK: - Properties
    static let shared = HTZNetWorkManager()
    private let provider = MoyaProvider<API>()
    private let reachabilityManager = NetworkReachabilityManager()
    
    // MARK: - Public Methods
    
    /// 统一网络请求方法
    /// - Parameters:
    ///   - target: API目标
    ///   - completion: 完成回调
    func request<T: HandyJSON>(_ target: API, 
                              completion: @escaping (Result<JSON, Error>) -> Void) {
        
        // 检查网络连通性
        guard reachabilityManager?.isReachable == true else {
            handleNetworkNotReachable(target: target, completion: completion)
            return
        }
        
        // 发起网络请求
        provider.request(target) { [weak self] result in
            switch result {
            case .success(let response):
                self?.handleSuccessResponse(response: response, completion: completion)
            case .failure(let error):
                self?.handleFailureResponse(error: error, target: target, completion: completion)
            }
        }
    }
    
    // MARK: - Private Methods
    
    private func handleNetworkNotReachable<T: HandyJSON>(target: API, 
                                                       completion: @escaping (Result<JSON, Error>) -> Void) {
        HTZLog("网络不可达，尝试使用缓存数据")
        
        // 尝试获取缓存数据
        if let cachedData = getCachedData(for: target) {
            completion(.success(JSON(cachedData)))
        } else if let mockData = MockDataManager.shared.getMockData(for: target) {
            completion(.success(JSON(mockData)))
        } else {
            completion(.failure(NetworkError.notConnected))
        }
    }
    
    private func handleSuccessResponse<T: HandyJSON>(response: Response, 
                                                   completion: @escaping (Result<JSON, Error>) -> Void) {
        do {
            let json = try JSON(response.data)
            
            // 记录成功日志
            HTZLog("网络请求成功: \(response.statusCode)")
            
            // 缓存响应数据
            cacheResponseData(response.data, for: response.request?.url?.absoluteString)
            
            completion(.success(json))
        } catch {
            HTZLog("JSON解析失败: \(error)")
            completion(.failure(NetworkError.jsonParsingFailed))
        }
    }
    
    private func handleFailureResponse<T: HandyJSON>(error: Error, 
                                                    target: API,
                                                    completion: @escaping (Result<JSON, Error>) -> Void) {
        HTZLog("网络请求失败: \(error)")
        
        // 尝试使用mock数据
        if let mockData = MockDataManager.shared.getMockData(for: target) {
            completion(.success(JSON(mockData)))
        } else {
            completion(.failure(error))
        }
    }
}

// MARK: - Network Error Definition
enum NetworkError: Error {
    case notConnected
    case jsonParsingFailed
    case invalidResponse
    case timeout
    
    var localizedDescription: String {
        switch self {
        case .notConnected:
            return "网络连接不可用"
        case .jsonParsingFailed:
            return "数据解析失败"
        case .invalidResponse:
            return "服务器响应无效"
        case .timeout:
            return "请求超时"
        }
    }
}
```

## 2. API接口定义

### API.swift
```swift
// 文件位置: htz/htz/Utils/API.swift
// 功能: 定义所有API接口和请求配置

import Moya

enum API {
    // 专辑相关
    case albums                                      // 获取所有专辑
    case sutras(page_index: NSInteger, page_size: NSInteger)           // 分页获取经典列表
    case items(sutra_id: String, page_index: NSInteger, page_size: NSInteger) // 获取专辑内容
    
    // 特定专辑
    case xingfuneixinchan                            // 性芬内省禅专辑
    case jingxinyangsheng                           // 净心养生专辑
    case mixinxiaoshipin                           // 密信小视频专辑
    
    // 用户相关
    case userInfo                                  // 获取用户信息
    case userLogin(phone: String, code: String)    // 用户登录
    
    // 播放相关
    case playHistory(page_index: NSInteger, page_size: NSInteger) // 播放历史
    case favoriteList(page_index: NSInteger, page_size: NSInteger) // 收藏列表
}

extension API: TargetType {
    var baseURL: URL {
        return URL(string: MoyaConfig.baseURL)!
    }
    
    var path: String {
        switch self {
        case .albums:
            return "albums.json"
        case .sutras:
            return "sutras.json"
        case .items:
            return "items.json"
        case .xingfuneixinchan:
            return "xingfuneixinchan/xingfuneixinchan.json"
        case .jingxinyangsheng:
            return "jingxinyangsheng/jingxinyangsheng.json"
        case .mixinxiaoshipin:
            return "mixinxiaoshipin/mixinxiaoshipin.json"
        case .userInfo:
            return "user/info"
        case .userLogin:
            return "user/login"
        case .playHistory:
            return "user/history"
        case .favoriteList:
            return "user/favorites"
        }
    }
    
    var method: Moya.Method {
        switch self {
        case .userLogin:
            return .post
        default:
            return .get
        }
    }
    
    var parameters: [String: Any]? {
        switch self {
        case .sutras(let page_index, let page_size):
            return ["page_index": page_index, "page_size": page_size]
        case .items(let sutra_id, let page_index, let page_size):
            return ["sutra_id": sutra_id, "page_index": page_index, "page_size": page_size]
        case .userLogin(let phone, let code):
            return ["phone": phone, "code": code]
        case .playHistory(let page_index, let page_size):
            return ["page_index": page_index, "page_size": page_size]
        case .favoriteList(let page_index, let page_size):
            return ["page_index": page_index, "page_size": page_size]
        default:
            return nil
        }
    }
    
    var parameterEncoding: ParameterEncoding {
        switch method {
        case .post:
            return JSONEncoding.default
        default:
            return URLEncoding.default
        }
    }
    
    var sampleData: Data {
        return "{}".data(using: .utf8)!
    }
    
    var headers: [String: String]? {
        return [
            "Content-Type": "application/json",
            "Accept": "application/json"
        ]
    }
}
```

```swift
import HandyJSON

struct HTZSutraInfoModel: HandyJSON {
    var id: String = ""
    var name: String = ""
    var description: String = ""
    var cover_url: String = ""
    var audio_count: Int = 0
    var total_duration: String = "00:00"
    var created_at: String = ""
    var updated_at: String = ""
    
    // MARK: - Computed Properties
    
    /// 完整的封面URL
    var fullCoverURL: String {
        return cover_url.hasPrefix("http") ? cover_url : "\(MoyaConfig.ossURL)\(cover_url)"
    }
    
    /// 格式化的专辑时长
    var formattedDuration: String {
        return total_duration.isEmpty ? "00:00" : total_duration
    }
    
    // MARK: - HandyJSON
    
    init() {}
    
    mutating func didFinishMapping() {
        // 数据映射完成后的处理
        if id.isEmpty {
            id = UUID().uuidString
        }
    }
}
```

### HTZMusicModel.swift
```swift
// 文件位置: htz/htz/Class/Play/Model/HTZMusicModel.swift
// 功能: 音频播放数据模型

import UIKit
import HandyJSON

class HTZMusicModel: NSObject, HandyJSON, NSCoding {
    
    // MARK: - Properties
    var id: String = ""
    var title: String = ""
    var artist: String = ""
    var audioUrl: String = ""
    var coverUrl: String = ""
    var duration: String = "00:00"
    var lyrics: String = ""
    var fileSize: Int64 = 0
    
    // MARK: - Playing State
    var isPlaying: Bool = false
    var currentTime: TimeInterval = 0
    var isDownloaded: Bool = false
    var localPath: String = ""
    
    // MARK: - Computed Properties
    
    /// 完整的音频URL
    var fullAudioURL: String {
        if isDownloaded && !localPath.isEmpty {
            return localPath
        }
        return audioUrl.hasPrefix("http") ? audioUrl : "\(MoyaConfig.ossURL)\(audioUrl)"
    }
    
    /// 完整的封面URL
    var fullCoverURL: String {
        return coverUrl.hasPrefix("http") ? coverUrl : "\(MoyaConfig.ossURL)\(coverUrl)"
    }
    
    /// 格式化的文件大小
    var formattedFileSize: String {
        return ByteCountFormatter.string(fromByteCount: fileSize, countStyle: .file)
    }
    
    // MARK: - Initialization
    
    override init() {
        super.init()
    }
    
    convenience init(model: HTZSutraItemModel) {
        self.init()
        self.id = model.id
        self.title = model.title
        self.audioUrl = model.fullAudioURL
        self.coverUrl = model.fullCoverURL
        self.duration = model.duration
        
        // 检查本地下载状态
        self.isDownloaded = DownloadManager.shared.isFileDownloaded(for: id)
        if isDownloaded {
            self.localPath = DownloadManager.shared.localPath(for: id)
        }
    }
    
    // MARK: - HandyJSON
    required init?(map: Map) {}
    
    func mapping(map: Map) {
        id <- map["id"]
        title <- map["title"]
        artist <- map["artist"]
        audioUrl <- map["audio_url"]
        coverUrl <- map["cover_url"]
        duration <- map["duration"]
        lyrics <- map["lyrics"]
        fileSize <- map["file_size"]
    }
    
    // MARK: - NSCoding
    func encode(with coder: NSCoder) {
        coder.encode(id, forKey: "id")
        coder.encode(title, forKey: "title")
        coder.encode(artist, forKey: "artist")
        coder.encode(audioUrl, forKey: "audioUrl")
        coder.encode(coverUrl, forKey: "coverUrl")
        coder.encode(duration, forKey: "duration")
        coder.encode(lyrics, forKey: "lyrics")
        coder.encode(fileSize, forKey: "fileSize")
        coder.encode(isDownloaded, forKey: "isDownloaded")
        coder.encode(localPath, forKey: "localPath")
    }
    
    required convenience init?(coder: NSCoder) {
        self.init()
        self.id = coder.decodeObject(forKey: "id") as? String ?? ""
        self.title = coder.decodeObject(forKey: "title") as? String ?? ""
        self.artist = coder.decodeObject(forKey: "artist") as? String ?? ""
        self.audioUrl = coder.decodeObject(forKey: "audioUrl") as? String ?? ""
        self.coverUrl = coder.decodeObject(forKey: "coverUrl") as? String ?? ""
        self.duration = coder.decodeObject(forKey: "duration") as? String ?? "00:00"
        self.lyrics = coder.decodeObject(forKey: "lyrics") as? String ?? ""
        self.fileSize = coder.decodeInt64(forKey: "fileSize")
        self.isDownloaded = coder.decodeBool(forKey: "isDownloaded")
        self.localPath = coder.decodeObject(forKey: "localPath") as? String ?? ""
    }
}
```

```swift
import Foundation

    static let isDebugMode = true
    static let enableNetworkLog = true
    #else
    static let isDebugMode = false
    static let enableNetworkLog = false
    #endif

    // MARK: - Helper Methods

        return !isDebugMode
```
