### 1.2 全局状态管理器实现

```swift
// HTZStateManager.swift
import Foundation
import RxSwift

class HTZStateManager {
    
    // MARK: - Singleton
    static let shared = HTZStateManager()
    private init() {}
    
    // MARK: - Properties
    private let disposeBag = DisposeBag()
    
    // MARK: - User State
    private let _userState = BehaviorSubject<UserState>(value: UserState())
    var userState: Observable<UserState> {
        return _userState.asObservable()
    }
    
    // MARK: - Player State
    private let _playerState = BehaviorSubject<PlayerState>(value: PlayerState())
    var playerState: Observable<PlayerState> {
        return _playerState.asObservable()
    }
    
    // MARK: - Network State
    private let _networkState = BehaviorSubject<NetworkState>(value: NetworkState())
    var networkState: Observable<NetworkState> {
        return _networkState.asObservable()
    }
    
    // MARK: - State Update Methods
    func updateUserState(_ state: UserState) {
        _userState.onNext(state)
    }
    
    func updatePlayerState(_ state: PlayerState) {
        _playerState.onNext(state)
    }
    
    func updateNetworkState(_ state: NetworkState) {
        _networkState.onNext(state)
    }
    
    // MARK: - State Persistence
    func saveState() {
        let stateData = StateManagerPersistence.shared.encodeStates(
            userState: try? _userState.value(),
            playerState: try? _playerState.value(),
            networkState: try? _networkState.value()
        )
        
        StateManagerPersistence.shared.saveStates(stateData)
    }
    
    func restoreState() {
        if let states = StateManagerPersistence.shared.loadStates() {
            if let userState = states.userState {
                _userState.onNext(userState)
            }
            if let playerState = states.playerState {
                _playerState.onNext(playerState)
            }
            if let networkState = states.networkState {
                _networkState.onNext(networkState)
            }
        }
    }
}

// MARK: - State Models
struct UserState {
    var isLoggedIn: Bool = false
    var userInfo: HTZUserInfoModel?
    var permissions: [String] = []
    var loginTime: Date?
    var sessionToken: String?
    
    var hasPermission(_ permission: String) -> Bool {
        return permissions.contains(permission)
    }
    
    var isSessionValid: Bool {
        guard let loginTime = loginTime else { return false }
        let sessionDuration = Date().timeIntervalSince(loginTime)
        return sessionDuration < 24 * 60 * 60 // 24小时
    }
}

struct PlayerState {
    var currentMusic: HTZMusicModel?
    var isPlaying: Bool = false
    var currentTime: TimeInterval = 0
    var duration: TimeInterval = 0
    var playList: [HTZMusicModel] = []
    var currentIndex: Int = 0
    var playMode: PlayMode = .sequence
    var volume: Float = 1.0
    var isShuffled: Bool = false
    
    enum PlayMode {
        case sequence    // 顺序播放
        case repeatOne   // 单曲循环
        case repeatAll   // 列表循环
        case shuffle     // 随机播放
    }
    
    var nextIndex: Int? {
        switch playMode {
        case .sequence:
            return currentIndex + 1 < playList.count ? currentIndex + 1 : nil
        case .repeatAll:
            return (currentIndex + 1) % playList.count
        case .shuffle:
            return (currentIndex + 1) % playList.count
        case .repeatOne:
            return currentIndex
        }
    }
    
    var previousIndex: Int? {
        switch playMode {
        case .sequence:
            return currentIndex - 1 >= 0 ? currentIndex - 1 : nil
        case .repeatAll:
            return currentIndex - 1 >= 0 ? currentIndex - 1 : playList.count - 1
        case .shuffle:
            return currentIndex - 1 >= 0 ? currentIndex - 1 : playList.count - 1
        case .repeatOne:
            return currentIndex
        }
    }
}

struct NetworkState {
    var isConnected: Bool = false
    var connectionType: ConnectionType = .unknown
    var requestCount: Int = 0
    var lastRequestTime: Date?
    var totalDataUsage: Int64 = 0
    
    enum ConnectionType {
        case wifi
        case cellular
        case none
        case unknown
    }
    
    var canStreamHighQuality: Bool {
        switch connectionType {
        case .wifi:
            return true
        case .cellular:
            return false // 默认蜂窝网络不使用高质量
        case .none, .unknown:
            return false
        }
    }
}
```

```swift
// MARK: - View State Models
struct AlbumViewState {
    var albums: [HTZSutraInfoModel] = []
    var isLoading: Bool = false
    var loadingMore: Bool = false
    var hasMoreData: Bool = true
    var currentPage: Int = 1
    var error: Error?
    var selectedAlbum: HTZSutraInfoModel?
    var searchKeyword: String = ""
    var isSearching: Bool = false
    
    var shouldShowLoadingView: Bool {
        return isLoading && albums.isEmpty
    }
    
    var shouldShowLoadingMoreIndicator: Bool {
        return loadingMore && !albums.isEmpty
    }
    
    var shouldShowEmptyView: Bool {
        return !isLoading && albums.isEmpty && searchKeyword.isEmpty
    }
    
    var shouldShowSearchResultView: Bool {
        return !isSearching && !albums.isEmpty && !searchKeyword.isEmpty
    }
    
    mutating func appendAlbums(_ newAlbums: [HTZSutraInfoModel]) {
        albums.append(contentsOf: newAlbums)
        hasMoreData = newAlbums.count >= 20 // 假设每页20个
        currentPage += 1
    }
    
    mutating func refreshWithAlbums(_ newAlbums: [HTZSutraInfoModel]) {
        albums = newAlbums
        currentPage = 1
        hasMoreData = newAlbums.count >= 20
        error = nil
        isLoading = false
        loadingMore = false
    }
}

struct PlayerViewState {
    var currentMusic: HTZMusicModel?
    var playList: [HTZMusicModel] = []
    var isPlaying: Bool = false
    var currentTime: TimeInterval = 0
    var duration: TimeInterval = 0
    var playbackRate: Float = 1.0
    var volume: Float = 1.0
    var isBuffering: Bool = false
    var isDragging: Bool = false
    var showPlaylist: Bool = false
    var showLyrics: Bool = false
    
    var formattedCurrentTime: String {
        return formatTime(currentTime)
    }
    
    var formattedDuration: String {
        return formatTime(duration)
    }
    
    var formattedRemainingTime: String {
        let remainingTime = duration - currentTime
        return "-\(formatTime(remainingTime))"
    }
    
    var progress: Float {
        guard duration > 0 else { return 0 }
        return Float(currentTime / duration)
    }
    
    var bufferingProgress: Float {
        // 这里应该从播放器获取实际的缓冲进度
        return 0.8 // 示例值
    }
    
    private func formatTime(_ time: TimeInterval) -> String {
        let minutes = Int(time) / 60
        let seconds = Int(time) % 60
        return String(format: "%02d:%02d", minutes, seconds)
    }
}
```

## 2. 状态持久化机制

### 2.1 状态持久化管理器

```swift
// StateManagerPersistence.swift
import Foundation
import KeychainAccess
import CoreData

class StateManagerPersistence {
    
    // MARK: - Singleton
    static let shared = StateManagerPersistence()
    private init() {}
    
    // MARK: - Keychain
    private let keychain: Keychain = {
        return Keychain(service: Bundle.main.bundleIdentifier ?? "com.htz.app")
            .accessibility(.whenUnlockedThisDeviceOnly)
    }()
    
    // MARK: - State Encoding
    func encodeStates(
        userState: UserState?,
        playerState: PlayerState?,
        networkState: NetworkState?
    ) -> StateData {
        
        var stateData = StateData()
        
        if let userState = userState,
           let userData = try? JSONEncoder().encode(userState) {
            stateData.userData = userData
        }
        
        if let playerState = playerState,
           let playerData = try? JSONEncoder().encode(playerState) {
            stateData.playerData = playerData
        }
        
        if let networkState = networkState,
           let networkData = try? JSONEncoder().encode(networkState) {
            stateData.networkData = networkData
        }
        
        return stateData
    }
    
    // MARK: - State Decoding
    func decodeStates(_ stateData: StateData) -> (
        userState: UserState?,
        playerState: PlayerState?,
        networkState: NetworkState?
    ) {
        
        var userState: UserState?
        var playerState: PlayerState?
        var networkState: NetworkState?
        
        if let userData = stateData.userData {
            userState = try? JSONDecoder().decode(UserState.self, from: userData)
        }
        
        if let playerData = stateData.playerData {
            playerState = try? JSONDecoder().decode(PlayerState.self, from: playerData)
        }
        
        if let networkData = stateData.networkData {
            networkState = try? JSONDecoder().decode(NetworkState.self, from: networkData)
        }
        
        return (userState, playerState, networkState)
    }
    
    // MARK: - Save States
    func saveStates(_ stateData: StateData) {
        // 用户状态保存到 Keychain（安全敏感）
        if let userData = stateData.userData {
            try? keychain.set(userData, key: "userState")
        }
        
        // 播放状态保存到 UserDefaults（非敏感）
        if let playerData = stateData.playerData {
            UserDefaults.standard.set(playerData, forKey: "playerState")
        }
        
        // 网络状态保存到 UserDefaults
        if let networkData = stateData.networkData {
            UserDefaults.standard.set(networkData, forKey: "networkState")
        }
        
        // 保存时间戳
        UserDefaults.standard.set(Date(), forKey: "stateSaveTimestamp")
    }
    
    // MARK: - Load States
    func loadStates() -> (
        userState: UserState?,
        playerState: PlayerState?,
        networkState: NetworkState?
    )? {
        
        // 检查状态是否过期
        if let saveTimestamp = UserDefaults.standard.object(forKey: "stateSaveTimestamp") as? Date {
            let timeSinceLastSave = Date().timeIntervalSince(saveTimestamp)
            if timeSinceLastSave > 7 * 24 * 60 * 60 { // 7天过期
                return nil
            }
        } else {
            return nil
        }
        
        var stateData = StateData()
        
        // 从 Keychain 加载用户状态
        if let userData = try? keychain.getData("userState") {
            stateData.userData = userData
        }
        
        // 从 UserDefaults 加载播放状态
        if let playerData = UserDefaults.standard.data(forKey: "playerState") {
            stateData.playerData = playerData
        }
        
        // 从 UserDefaults 加载网络状态
        if let networkData = UserDefaults.standard.data(forKey: "networkState") {
            stateData.networkData = networkData
        }
        
        return decodeStates(stateData)
    }
    
    // MARK: - Clear States
    func clearStates() {
        // 清除 Keychain 中的敏感数据
        try? keychain.remove("userState")
        
        // 清除 UserDefaults 中的数据
        UserDefaults.standard.removeObject(forKey: "playerState")
        UserDefaults.standard.removeObject(forKey: "networkState")
        UserDefaults.standard.removeObject(forKey: "stateSaveTimestamp")
    }
}

// MARK: - State Data Model
struct StateData {
    var userData: Data?
    var playerData: Data?
    var networkData: Data?
}
```

## 3. 状态绑定和响应式更新

### 3.1 响应式状态绑定

```swift
// HTZStateBinding.swift
import Foundation
import RxSwift
import RxCocoa

extension UIViewController {
    
    func bindAlbumViewState(
        to viewModel: HTZAlbumViewModel,
        tableView: UITableView,
        loadingView: UIView?,
        emptyView: UIView?,
        refreshControl: UIRefreshControl?
    ) {
        let disposeBag = DisposeBag()
        
        // 绑定专辑数据
        HTZViewStateManager.shared.albumViewState
            .map { $0.albums }
            .bind(to: tableView.rx.items(cellIdentifier: "AlbumCell", cellType: HTZAlbumCell.self)) { index, model, cell in
                cell.configure(with: model)
            }
            .disposed(by: disposeBag)
        
        // 绑定加载状态
        HTZViewStateManager.shared.albumViewState
            .map { $0.shouldShowLoadingView }
            .bind(to: loadingView?.rx.isHidden ?? Observable.just(true))
            .disposed(by: disposeBag)
        
        // 绑定空数据状态
        HTZViewStateManager.shared.albumViewState
            .map { $0.shouldShowEmptyView }
            .bind(to: emptyView?.rx.isHidden ?? Observable.just(true))
            .disposed(by: disposeBag)
        
        // 绑定刷新控制
        if let refreshControl = refreshControl {
            HTZViewStateManager.shared.albumViewState
                .map { !$0.isLoading }
                .bind(to: refreshControl.rx.isRefreshing)
                .disposed(by: disposeBag)
        }
        
        // 绑定错误处理
        HTZViewStateManager.shared.albumViewState
            .compactMap { $0.error }
            .subscribe(onNext: { error in
                self.showError(error.localizedDescription)
            })
            .disposed(by: disposeBag)
    }
}

// MARK: - RxSwift Extensions
extension Reactive where Base: UIRefreshControl {
    var isRefreshing: Binder<Bool> {
        return Binder(self.base) { refreshControl, refreshing in
            if refreshing {
                refreshControl.beginRefreshing()
            } else {
                refreshControl.endRefreshing()
            }
        }
    }
}

extension Reactive where Base: UIView {
    var isHidden: Binder<Bool> {
        return Binder(self.base) { view, hidden in
            view.isHidden = hidden
        }
    }
}
```

### 3.2 状态变更通知机制

```swift
// HTZStateNotificationCenter.swift
import Foundation
import RxSwift

class HTZStateNotificationCenter {
    
    // MARK: - Singleton
    static let shared = HTZStateNotificationCenter()
    private init() {}
    
    // MARK: - Notification Subjects
    private let _userStateDidChange = PublishSubject<(UserState, UserState)>()
    private let _playerStateDidChange = PublishSubject<(PlayerState, PlayerState)>()
    private let _networkStateDidChange = PublishSubject<(NetworkState, NetworkState)>()
    
    // MARK: - Observable Properties
    var userStateDidChange: Observable<(UserState, UserState)> {
        return _userStateDidChange.asObservable()
    }
    
    var playerStateDidChange: Observable<(PlayerState, PlayerState)> {
        return _playerStateDidChange.asObservable()
    }
    
    var networkStateDidChange: Observable<(NetworkState, NetworkState)> {
        return _networkStateDidChange.asObservable()
    }
    
    // MARK: - Notification Methods
    func notifyUserStateChange(oldState: UserState, newState: UserState) {
        _userStateDidChange.onNext((oldState, newState))
    }
    
    func notifyPlayerStateChange(oldState: PlayerState, newState: PlayerState) {
        _playerStateDidChange.onNext((oldState, newState))
    }
    
    func notifyNetworkStateChange(oldState: NetworkState, newState: NetworkState) {
        _networkStateDidChange.onNext((oldState, newState))
    }
    
    // MARK: - Specific Notifications
    func notifyUserLogin(userInfo: HTZUserInfoModel) {
        let newState = UserState(
            isLoggedIn: true,
            userInfo: userInfo,
            loginTime: Date(),
            sessionToken: userInfo.token
        )
        
        HTZStateManager.shared.updateUserState(newState)
    }
    
    func notifyUserLogout() {
        let newState = UserState()
        HTZStateManager.shared.updateUserState(newState)
        
        // 清除本地状态
        StateManagerPersistence.shared.clearStates()
    }
    
    func notifyMusicPlay(music: HTZMusicModel) {
        var playerState = try? HTZStateManager.shared.playerState.value()
        playerState?.currentMusic = music
        playerState?.isPlaying = true
        
        if let newState = playerState {
            HTZStateManager.shared.updatePlayerState(newState)
        }
    }
    
    func notifyMusicPause() {
        var playerState = try? HTZStateManager.shared.playerState.value()
        playerState?.isPlaying = false
        
        if let newState = playerState {
            HTZStateManager.shared.updatePlayerState(newState)
        }
    }
    
    func notifyNetworkConnectionChanged(isConnected: Bool, type: NetworkState.ConnectionType) {
        let newState = NetworkState(
            isConnected: isConnected,
            connectionType: type
        )
        
        HTZStateManager.shared.updateNetworkState(newState)
    }
}
```
