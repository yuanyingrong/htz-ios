```swift
// 文件位置: htz/htz/Class/Album/ViewModel/HTZAlbumViewModel.swift

func requestData(completion: (() -> Void)?) {
    // 构建请求参数
    let params = [
        "page_index": pageIndex,
        "page_size": pageSize
    ]
    
    // 发起网络请求
    NetWorkRequest(API.sutras(page_index: pageIndex, page_size: pageSize)) { (result) in
        switch result {
        case .success(let json):
            // 解析JSON数据
            if let data = json["data"].array {
                var models: [HTZSutraInfoModel] = []
                for item in data {
                    if let model = HTZSutraInfoModel.deserialize(from: item.rawString()) {
                        models.append(model)
                    }
                }
                
                // 更新数据源
                if self.pageIndex == 1 {
                    self.sutraModels = models
                } else {
                    self.sutraModels.append(contentsOf: models)
                }
            }
        case .failure(let error):
            // 错误处理，可能使用mock数据
            HTZLog("专辑数据获取失败: \(error)")
        }
        
        completion?()
    }
}
```

## 4. 网络请求核心层

### 4.1 HTZNetWorkManager 统一请求方法
```swift
// 文件位置: htz/htz/Utils/HTZNetWorkManager.swift

func NetWorkRequest<T: HandyJSON>(_ target: API, completion: @escaping (Result<JSON, Error>) -> Void) {
    // 检查网络连通性
    guard NetworkReachabilityManager()!.isReachable else {
        // 网络不可达，使用mock数据或返回错误
        if let mockData = getMockData(for: target) {
            completion(.success(JSON(mockData)))
        } else {
            completion(.failure(NetworkError.notConnected))
        }
        return
    }
    
    // 使用Moya发起请求
    provider.request(target) { result in
        switch result {
        case .success(let response):
            do {
                let json = try JSON(response.data)
                completion(.success(json))
            } catch {
                completion(.failure(error))
            }
        case .failure(let error):
            // 请求失败，尝试使用mock数据
            if let mockData = self.getMockData(for: target) {
                completion(.success(JSON(mockData)))
            } else {
                completion(.failure(error))
            }
        }
    }
}
```

### 4.2 API 定义
```swift
// 文件位置: htz/htz/Utils/API.swift

enum API {
    case sutras(page_index: NSInteger, page_size: NSInteger)
    case items(sutra_id: String, page_index: NSInteger, page_size: NSInteger)
    case xingfuneixinchan
    case jingxinyangsheng
    case mixinxiaoshipin
}

extension API: TargetType {
    var baseURL: URL {
        return URL(string: MoyaConfig.baseURL)!
    }
    
    var path: String {
        switch self {
        case .sutras:
            return "sutras.json"
        case .items:
            return "items.json"
        case .xingfuneixinchan:
            return "xingfuneixinchan/xingfuneixinchan.json"
        case .jingxinyangsheng:
            return "jingxinyangsheng/jingxinyangsheng.json"
        case .mixinxiaoshipin:
            return "mixinxiaoshipin/mixinxiaoshipin.json"
        }
    }
    
    var method: Moya.Method {
        return .get
    }
    
    var parameters: [String: Any]? {
        switch self {
        case .sutras(let page_index, let page_size):
            return ["page_index": page_index, "page_size": page_size]
        case .items(let sutra_id, let page_index, let page_size):
            return ["sutra_id": sutra_id, "page_index": page_index, "page_size": page_size]
        default:
            return nil
        }
    }
}
```

## 6. 错误处理和容错机制

### 6.1 网络请求失败处理
1. **网络不可达**: 显示网络提示，使用本地缓存或mock数据
2. **服务器错误**: 记录错误日志，返回默认数据
3. **数据解析错误**: 跳过错误数据，继续解析其他数据
4. **UI更新**: 主线程更新UI，结束加载状态

### 6.2 Mock数据机制
```swift
// 在网络请求失败时提供默认数据
func getMockData(for target: API) -> [String: Any]? {
    switch target {
    case .sutras:
        return ["data": mockSutrasData()]
    case .xingfuneixinchan:
        return ["data": mockXingfuneixinchanData()]
    default:
        return nil
    }
}
```

这个调用链路展示了从用户操作到网络请求再到数据展示的完整流程，体现了清晰的分层架构和良好的错误处理机制。
