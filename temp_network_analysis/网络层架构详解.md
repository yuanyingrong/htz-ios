```swift
private func createProvider() -> MoyaProvider<API> {
    // 自定义会话配置
    let configuration = URLSessionConfiguration.default
    configuration.timeoutIntervalForRequest = 30
    configuration.timeoutIntervalForResource = 60
    configuration.requestCachePolicy = .reloadIgnoringLocalCacheData
    
    // 设置网络服务类型
    configuration.networkServiceType = .responsiveAV
    
    // 自定义会话代理
    let session = Session(configuration: configuration)
    
    // 插件配置
    let plugins: [PluginType] = [
        // 网络活动指示器插件
        NetworkActivityPlugin { change, _ in
            DispatchQueue.main.async {
                UIApplication.shared.isNetworkActivityIndicatorVisible = 
                (change == .began)
            }
        },
        
        // 自定义日志插件
        CustomNetworkLoggerPlugin(),
        
        // 缓存插件
        NetworkCachePlugin(),
        
        // 重试插件
        RetryPlugin(count: 3),
        
        // 认证插件
        AccessTokenPlugin { request in
            return HTZUserManager.shared.accessToken ?? ""
        }
    ]
    
    return MoyaProvider<API>(session: session, plugins: plugins)
}
```

```swift
// CustomNetworkLoggerPlugin.swift
import Moya
import SwiftyJSON

class CustomNetworkLoggerPlugin: PluginType {
    
    private let dateFormatter: DateFormatter = {
        let formatter = DateFormatter()
        formatter.dateFormat = "yyyy-MM-dd HH:mm:ss.SSS"
        return formatter
    }()
    
    func willSend(_ request: RequestType, target: TargetType) {
        let timestamp = dateFormatter.string(from: Date())
        
        var logMessage = "\n🚀 [\(timestamp)] 发起网络请求\n"
        logMessage += "┌─ URL: \(request.url?.absoluteString ?? "unknown")\n"
        logMessage += "├─ Method: \(request.httpMethod ?? "GET")\n"
        logMessage += "├─ Headers: \(request.allHTTPHeaderFields?.description ?? "none")\n"
        
        if let bodyData = request.httpBody,
           let bodyString = String(data: bodyData, encoding: .utf8) {
            logMessage += "└─ Body: \(bodyString)\n"
        } else {
            logMessage += "└─ Body: none\n"
        }
        
        HTZLog(logMessage, level: .network)
    }
    
    func didReceive(_ result: Result<Response, MoyaError>, target: TargetType) {
        let timestamp = dateFormatter.string(from: Date())
        
        switch result {
        case .success(let response):
            var logMessage = "\n✅ [\(timestamp)] 网络请求成功\n"
            logMessage += "┌─ Status Code: \(response.statusCode)\n"
            logMessage += "├─ URL: \(response.request?.url?.absoluteString ?? "unknown")\n"
            logMessage += "├─ Data Size: \(response.data.count) bytes\n"
            
            do {
                let json = try JSON(response.data)
                if let jsonString = json.rawString() {
                    logMessage += "└─ Response: \(jsonString)\n"
                }
            } catch {
                logMessage += "└─ Response: [无法解析为JSON]\n"
            }
            
            HTZLog(logMessage, level: .network)
            
        case .failure(let error):
            var logMessage = "\n❌ [\(timestamp)] 网络请求失败\n"
            logMessage += "├─ Error: \(error.localizedDescription)\n"
            logMessage += "└─ Target: \(target)\n"
            
            HTZLog(logMessage, level: .network)
        }
    }
}
```

### 2.3 网络缓存插件实现

```swift
// NetworkCachePlugin.swift
import Moya
import SwiftyJSON

class NetworkCachePlugin: PluginType {
    
    private let cacheManager = NetworkCacheManager.shared
    
    func willSend(_ request: RequestType, target: TargetType) {
        // 检查是否有有效缓存
        if let cachedResponse = cacheManager.getCachedResponse(for: target) {
            HTZLog("使用缓存数据: \(target)", level: .cache)
            
            // 将缓存数据转换为响应
            let statusCode = cachedResponse["status_code"].int ?? 200
            let dataString = cachedResponse["data"].stringValue
            
            if let responseData = dataString.data(using: .utf8) {
                // 创建缓存响应
                let cachedResponse = Response(
                    statusCode: statusCode,
                    data: responseData,
                    request: request.request,
                    response: nil
                )
                
                // 注入缓存响应
                // 注意：这里需要通过其他方式来使用缓存响应
                // 因为Moya插件本身不能直接替换响应
            }
        }
    }
    
    func didReceive(_ result: Result<Response, MoyaError>, target: TargetType) {
        switch result {
        case .success(let response):
            // 只缓存GET请求的成功响应
            if target.method == .get && response.statusCode == 200 {
                cacheManager.cacheResponse(response, for: target)
            }
        case .failure:
            break
        }
    }
}

// NetworkCacheManager.swift
class NetworkCacheManager {
    
    static let shared = NetworkCacheManager()
    private init() {}
    
    private let userDefaults = UserDefaults.standard
    private let cacheKeyPrefix = "NetworkCache_"
    
    func getCachedResponse(for target: TargetType) -> JSON? {
        let key = cacheKeyPrefix + target.path
        
        guard let cachedData = userDefaults.data(forKey: key),
              let json = try? JSON(data: cachedData) else {
            return nil
        }
        
        // 检查缓存是否过期
        if let timestamp = json["timestamp"].double {
            let cacheTime = Date(timeIntervalSince1970: timestamp)
            let expirationTime = cacheTime.addingTimeInterval(MoyaConfig.cacheExpirationTime)
            
            if Date() > expirationTime {
                userDefaults.removeObject(forKey: key)
                return nil
            }
        }
        
        return json
    }
    
    func cacheResponse(_ response: Response, for target: TargetType) {
        do {
            let responseJSON = try JSON(response.data)
            var cacheData: [String: Any] = [
                "status_code": response.statusCode,
                "timestamp": Date().timeIntervalSince1970,
                "data": responseJSON.rawString() ?? "{}"
            ]
            
            let key = cacheKeyPrefix + target.path
            let cachedData = try JSONSerialization.data(withJSONObject: cacheData)
            userDefaults.set(cachedData, forKey: key)
            
            HTZLog("缓存响应数据: \(target)", level: .cache)
        } catch {
            HTZLog("缓存数据失败: \(error)", level: .error)
        }
    }
    
    func clearCache() {
        let domains = userDefaults.dictionaryRepresentation()
        domains.keys.forEach { key in
            if key.hasPrefix(cacheKeyPrefix) {
                userDefaults.removeObject(forKey: key)
            }
        }
        HTZLog("清除所有网络缓存", level: .cache)
    }
}
```

### 2.4 重试机制插件

```swift
// RetryPlugin.swift
import Moya

class RetryPlugin: PluginType {
    
    private let retryCount: Int
    private let retryDelay: TimeInterval
    
    init(count: Int = 3, delay: TimeInterval = 1.0) {
        self.retryCount = count
        self.retryDelay = delay
    }
    
    func process(_ result: Result<Response, MoyaError>, target: TargetType) -> Result<Response, MoyaError> {
        if case .failure(let error) = result {
            if shouldRetry(for: error, target: target, retryCount: retryCount) {
                return .failure(error) // 将在provider的重试逻辑中处理
            }
        }
        return result
    }
    
    private func shouldRetry(for error: MoyaError, target: TargetType, retryCount: Int) -> Bool {
        guard retryCount > 0 else { return false }
        
        switch error {
        case .underlying(let nsError, _):
            let errorCode = (nsError as NSError).code
            
            // 重试条件：超时、服务器错误、网关错误等
            switch errorCode {
            case NSURLErrorTimedOut,      // 请求超时
                 NSURLErrorCannotConnectToHost,  // 无法连接主机
                 NSURLErrorNetworkConnectionLost, // 网络连接丢失
                 NSURLErrorNotConnectedToInternet: // 无网络连接
                return true
                
            case 500...599: // 服务器错误
                return true
                
            default:
                return false
            }
            
        default:
            return false
        }
    }
}
```

```swift
// AccessTokenPlugin.swift
import Moya

class AccessTokenPlugin: PluginType {
    
    private let tokenClosure: () -> String?
    
    init(tokenClosure: @escaping () -> String?) {
        self.tokenClosure = tokenClosure
    }
    
    func prepare(_ request: URLRequest, target: TargetType) -> URLRequest {
        guard let token = tokenClosure(),
              !token.isEmpty,
              target.path.contains("user/") || target.path.contains("play/") else {
            return request
        }
        
        var request = request
        request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
        return request
    }
}
```

### 3.2 错误处理装饰器

```swift
// ErrorHandlingDecorator.swift
import Moya

class ErrorHandlingDecorator {
    
    static func decorate<T: HandyJSON>(
        _ target: API,
        completion: @escaping (Result<T, Error>) -> Void
    ) -> (Result<JSON, Error>) -> Void {
        return { result in
            switch result {
            case .success(let json):
                self.handleJSONResponse(json, completion: completion)
            case .failure(let error):
                self.handleNetworkError(error, completion: completion)
            }
        }
    }
    
    private static func handleJSONResponse<T: HandyJSON>(
        _ json: JSON,
        completion: @escaping (Result<T, Error>) -> Void
    ) {
        // 检查业务状态码
        if let code = json["code"].int, code != 200 {
            let message = json["message"].string ?? "未知错误"
            let businessError = NetworkError.businessError(code: code, message: message)
            completion(.failure(businessError))
            return
        }
        
        // 解析数据模型
        if let dataString = json["data"].rawString(),
           let model = T.deserialize(from: dataString) {
            completion(.success(model))
        } else {
            completion(.failure(NetworkError.jsonParsingFailed))
        }
    }
    
    private static func handleNetworkError<T: HandyJSON>(
        _ error: Error,
        completion: @escaping (Result<T, Error>) -> Void
    ) {
        guard let moyaError = error as? MoyaError else {
            completion(.failure(error))
            return
        }
        
        let networkError: NetworkError
        switch moyaError {
        case .underlying(let nsError, _):
            networkError = self.mapURLError(nsError as NSError)
        case .imageMapping, .jsonMapping, .stringMapping, .objectMapping, .encodableMapping:
            networkError = NetworkError.jsonParsingFailed
        case .statusCode(let response):
            networkError = self.mapStatusCode(response.statusCode)
        case .requestMapping, .parameterEncoding:
            networkError = NetworkError.invalidResponse
        }
        
        completion(.failure(networkError))
    }
    
    private static func mapURLError(_ error: NSError) -> NetworkError {
        switch error.code {
        case NSURLErrorNotConnectedToInternet:
            return .notConnected
        case NSURLErrorTimedOut:
            return .timeout
        default:
            return NetworkError(rawValue: error.code) ?? .invalidResponse
        }
    }
    
    private static func mapStatusCode(_ statusCode: Int) -> NetworkError {
        switch statusCode {
        case 401:
            return .unauthorized
        case 403:
            return .forbidden
        case 500...599:
            return .serverError(code: statusCode)
        default:
            return .invalidResponse
        }
    }
}
```

### 4.1 请求去重机制

```swift
// RequestDeduplicator.swift
import Foundation
import Moya

class RequestDeduplicator {
    
    static let shared = RequestDeduplicator()
    private init() {}
    
    private var ongoingRequests: [String: [(Result<JSON, Error>) -> Void]] = [:]
    private let queue = DispatchQueue(label: "RequestDeduplicator")
    
    func deduplicate<T: HandyJSON>(
        target: API,
        request: (@escaping (Result<JSON, Error>) -> Void) -> Void,
        completion: @escaping (Result<T, Error>) -> Void
    ) {
        let key = requestKey(for: target)
        
        queue.async { [weak self] in
            guard let self = self else { return }
            
            if var handlers = self.ongoingRequests[key] {
                // 请求已存在，添加到处理器队列
                handlers.append({ result in
                    ErrorHandlingDecorator.decorate(target)(result)(completion)
                })
                self.ongoingRequests[key] = handlers
            } else {
                // 创建新请求
                self.ongoingRequests[key] = [{ result in
                    ErrorHandlingDecorator.decorate(target)(result)(completion)
                }]
                
                // 执行实际请求
                request { [weak self] result in
                    self?.handleResult(key: key, result: result)
                }
            }
        }
    }
    
    private func handleResult(key: String, result: Result<JSON, Error>) {
        queue.async { [weak self] in
            guard let self = self else { return }
            
            if let handlers = self.ongoingRequests[key] {
                DispatchQueue.main.async {
                    handlers.forEach { $0(result) }
                }
            }
            
            self.ongoingRequests.removeValue(forKey: key)
        }
    }
    
    private func requestKey(for target: API) -> String {
        return "\(target.path)_\(target.parameters?.description ?? "")"
    }
}
```

### 4.2 请求池管理

```swift
// RequestPoolManager.swift
import Foundation
import Dispatch

class RequestPoolManager {
    
    static let shared = RequestPoolManager()
    private init() {}
    
    private let maxConcurrentRequests = 5
    private let semaphore = DispatchSemaphore(value: 5)
    private let queue = DispatchQueue(label: "RequestPool")
    
    func execute<T: HandyJSON>(
        _ operation: @escaping (@escaping (Result<T, Error>) -> Void) -> Void,
        completion: @escaping (Result<T, Error>) -> Void
    ) {
        queue.async { [weak self] in
            guard let self = self else { return }
            
            self.semaphore.wait()
            
            DispatchQueue.global().async {
                operation { result in
                    completion(result)
                    self.semaphore.signal()
                }
            }
        }
    }
}
```
